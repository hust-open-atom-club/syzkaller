# Read from or write to an aligned address in the first 4 KiB of the mapped
# VCPU stack page. The base must match LOONG64_ADDR_STACK_BASE in executor/kvm.h.
# Read results are stored in guest CSR KS0 (0x30), which is observable through
# KVM_GET_ONE_REG.
loong64_syzos_memop_len = 1, 2, 4, 8

syzos_api_memop$loong64 {
	base	const[0x80024000, int64]
	offset	int64[0:4088, 8]
	value	int64
	len	flags[loong64_syzos_memop_len, int64]
	op	int64[0:1]
}

	any	int64
]

# Hardware-backed CSRs listed by kvm_init_gcsr_flag() and used by the current
# one-reg seeds. Arbitrary 14-bit values remain available through @any.
